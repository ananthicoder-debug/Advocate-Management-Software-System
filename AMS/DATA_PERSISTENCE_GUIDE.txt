/*
 * Location: src/com/ams/util/DBConnection.java
 * 
 * KEY FEATURES:
 * ✓ Attempts Oracle connection silently (no error dialogs)
 * ✓ Automatic fallback to demo mode
 * ✓ Connection caching for performance
 * ✓ No console spam with error messages
 * 
 * CONNECTION LOGIC:
 * 1. Try primary URL: jdbc:oracle:thin:@//localhost:1521/XE
 * 2. Fallback URL:    jdbc:oracle:thin:@localhost:1521:XE
 * 3. Both fail? → Return null, trigger demo mode
 * 
 * METHODS:
 * - getConnection()    : Returns Connection or null
 * - testConnection()   : Returns boolean (no exceptions)
 * - getLastError()     : Gets error message string
 * - setAutoCommit()    : Sets commit mode if connected
 */

// 5. DATA PERSISTENCE IN ORACLE DATABASE
// ============================================================

/*
 * WHEN ORACLE DATABASE IS AVAILABLE:
 * 
 * All data automatically saves to these tables:
 * 
 * TABLE: CASES
 * ├─ case_id (PK)
 * ├─ c_id (client ID)
 * ├─ c_title (case title) ← Data entered by user
 * ├─ c_type (CIVIL/CRIMINAL/FAMILY)
 * ├─ status (PENDING/ACTIVE/CLOSED/ARCHIVED)
 * ├─ priority_level (1-3)
 * ├─ filed_date (auto: today)
 * ├─ court_name
 * └─ fee_amount
 * 
 * TABLE: HEARING
 * ├─ h_id (PK)
 * ├─ case_id (FK → CASES)
 * ├─ h_date
 * ├─ h_time
 * ├─ court_house
 * ├─ room_no
 * ├─ purpose
 * └─ status (UPCOMING/COMPLETED/ADJOURNED)
 * 
 * TABLE: REMINDER
 * ├─ rem_id (PK)
 * ├─ a_id (advocate ID)
 * ├─ case_id (FK → CASES)
 * ├─ due_date
 * ├─ message
 * ├─ priority (HIGH/MEDIUM/LOW)
 * └─ rem_status (PENDING/COMPLETED/DISMISSED)
 * 
 * TABLE: EVIDENCE
 * ├─ e_id (PK)
 * ├─ case_id (FK → CASES)
 * ├─ e_type (DOCUMENT/PHOTO/WITNESS)
 * ├─ e_source
 * ├─ admissibility (ADMITTED/PENDING/REJECTED)
 * └─ verified_by (advocate ID)
 * 
 * TABLE: COMMUNICATION
 * ├─ com_id (PK)
 * ├─ c_id (client ID)
 * ├─ case_id (FK → CASES)
 * ├─ a_id (advocate ID)
 * ├─ init_date (auto: now)
 * ├─ comm_mode (CALL/EMAIL/MEETING/MESSAGE)
 * ├─ summary
 * └─ follow_action
 * 
 * TABLE: TIMELINE
 * ├─ time_id (PK)
 * ├─ case_id (FK → CASES)
 * ├─ entry_date
 * ├─ title
 * ├─ description
 * ├─ next_step
 * └─ status_indicator (ON_TRACK/DELAYED/COMPLETED)
 */

 * SELECT COUNT(*) FROM REMINDER;
 * 
 * // View recent additions (last 5 cases)
 * SELECT * FROM CASES 
 * WHERE ROWNUM <= 5 
 * ORDER BY case_id DESC;

/*
 * Example 1: CasesPage.java - Add and Display Cases
 * ─────────────────────────────────────────────────
 * 
 * Location: src/com/ams/advocate/CasesPage.java
 * 
 * Key methods:
 * - showAddCaseDialog()     : Opens form, saves to DB or Demo
 * - loadCases()             : Called after add, refreshes table
 * - addDemoCases()          : Loads from DemoDataStore
 * 
 * Persistence flow:
 * 1. User clicks "+ Add Case"
 * 2. Dialog opens with form fields
 * 3. User enters: Title, Type, Court, Client ID, Status, Priority
 * 4. Clicks "Save Case"
 * 5. If DB available: INSERT into CASES + ADVOCATE_CASE tables
 * 6. If no DB: DemoDataStore.addCaseToDemo() stores in memory
 * 7. Success message shown
 * 8. loadCases() called to refresh display
 * 9. New case appears in table immediately
 * 
 * Code snippet:
 * ────────────
 * if (con != null) {
 *     // Production: Write to Oracle DB
 *     PreparedStatement ps = con.prepareStatement(
 *         "INSERT INTO CASES(...) VALUES(...)");
 *     ps.executeUpdate();
 *     con.commit();
 * } else {
 *     // Demo: Write to in-memory store
 *     DemoDataStore.addCaseToDemo(
 *         id, title, client, type, status, 
 *         priority, date, court);
 * }
 */
